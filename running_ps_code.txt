# 1. Clone the repository
Write-Host "Cloning repository..." -ForegroundColor Cyan
git clone https://github.com/chittranshi622-rgb/portfolio.git
Set-Location portfolio

# 2. Install Backend Dependencies
Write-Host "Installing backend dependencies..." -ForegroundColor Cyan
Set-Location backend
npm install
Set-Location ..

# 3. Install Frontend Dependencies
Write-Host "Installing frontend dependencies..." -ForegroundColor Cyan
Set-Location frontend
npm install
Set-Location ..

# 4. Create default backend .env file if missing
if (-not (Test-Path "backend/.env")) {
    "PORT=5000`nDB_HOST=localhost`nDB_USER=root`nDB_PASS=" | Out-File -FilePath "backend/.env" -Encoding utf8
}

# 5. Start Backend Server on Port 5000 in a new window
Write-Host "Starting Backend server..." -ForegroundColor Green
Start-Process powershell -ArgumentList "-NoExit", "-Command", "Set-Location backend; node server.js"

# 6. Start Frontend Server on Port 5173 in a new window
Write-Host "Starting Frontend server..." -ForegroundColor Green
Start-Process powershell -ArgumentList "-NoExit", "-Command", "Set-Location frontend; npm run dev"

# 7. Wait briefly for servers to start, then open portfolio in browser
Start-Sleep -Seconds 3
Start-Process "http://localhost:5173"

Write-Host "Portfolio launched successfully at http://localhost:5173!" -ForegroundColor Green
